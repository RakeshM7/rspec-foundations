require 'rspec/core/rake_task'
require 'allure-rspec'

# ── Test tasks ───────────────────────────────
namespace :rspec do
  desc 'Run smoke tests only (fast, <5 min — run on every PR)'
  RSpec::Core::RakeTask.new(:smoke) do |t|
    t.rspec_opts = '--tag smoke'
  end

  desc 'Run full regression suite (slow — nightly CI only)'
  RSpec::Core::RakeTask.new(:regression) do |t|
    t.rspec_opts = '--tag regression'
  end

  desc 'Run UI tests only'
  RSpec::Core::RakeTask.new(:ui) do |t|
    t.pattern = 'spec/ui/**/*_spec.rb'
  end

  desc 'Run API tests only (no browser needed)'
  RSpec::Core::RakeTask.new(:api) do |t|
    t.pattern = 'spec/api/**/*_spec.rb'
  end

  desc 'Run tests for a specific TPCR entity: rake rspec:entity[tickets]'
  RSpec::Core::RakeTask.new(:entity, :name) do |t, args|
    t.pattern = "spec/**/#{args[:name]}/**/*_spec.rb"
  end
end

# ── Allure tasks ─────────────────────────────
namespace :allure do
  desc 'Generate HTML report from allure-results/'
  task :generate do
    sh 'allure generate allure-results -o allure-report --clean'
    puts 'Report generated at allure-report/index.html'
  end

  desc 'Open Allure report in browser (generates first if needed)'
  task open: :generate do
    sh 'allure open allure-report'
  end
end

# ── Cleanup task ─────────────────────────────

task default: 'rspec:smoke'
